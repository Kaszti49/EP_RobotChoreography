// =====================================================================
//  K3 -- PWM -> SEBESSEG JELLEGGORBE, KEREKENKENT ES IRANYONKENT
//  Show-koreografia: 4. meres a hat robotbol mindegyiken
// =====================================================================
//
//  MIT MER?
//    Hany impulzus/masodperccel forog a kerek adott PWM-nel. Nyolc PWM-
//    erteken meri, kulon a bal es a jobb kereken, elore es hatra is.
//    A vegen a program MAGA illeszt egyenest a pontokra, es kiirja a ket
//    szamot, ami a show-vezerlesbe kell.
//
//  MI EZ A KET SZAM?
//    A motor jo kozelitessel linearis:   PWM = HOLTSAV + KFF * sebesseg
//      HOLTSAV : ahol elindul (a K0 ezt mar merte -- itt ellenorzod)
//      KFF     : mennyi PWM kell 1 impulzus/s sebesseghez
//    Ez a FEED-FORWARD: a vezerles ebbol tudja, MENNYIT kell adnia, mielott
//    barmilyen hibat latna. A PD-nek igy csak a maradek par szazalekot kell
//    korrigalnia -- ettol lesz sima es egyforma a hat robot mozgasa.
//    Feed-forward nelkul a PD egyedul is mukodne, de mindig kesne, es
//    minden robot maskepp kesne.
//
//  ELOKESZULET
//    - A SHOW PADLOJAN, kb. 2 m szabad hely elore (a robot elindul!).
//    - Friss/teli akku. A jelleggorbe akkufeszultseg-fuggo -- ugyanazzal az
//      akku-allapottal merj, amivel a show megy majd.
//    - A robot a meres alatt kanyarodhat, ez normalis (nincs szabalyzas).
//
//  HASZNALAT
//    Start -> kb. 8 mp elore, 2 mp szunet, kb. 8 mp hatra -> eredmeny.
// =====================================================================

// --------------------- beallitasok ---------------------
//  HELYIGENY -- ezt szamold ki, mielott rovidited a listat!
//  A robot a teljes sopres alatt megteszi kb.
//     osszeg( (PWM - holtsav)/KFF * mm_per_imp * (BEALL_MS+MERES_MS) )
//  utat. A 8 pontos listaval ez TOBB MINT 5 METER elore, es ugyanennyi hatra.
//  Ha nincs ennyi helyed, ROVIDITSD A LISTAT -- ne hagyd, hogy a robot
//  nekimenjen valaminek a sopres kozben.
//
//  (2026-09-07: a 6 pontos futas nagy PWM-en falba/kezbe utkozott. Emiatt a
//   bal kerek "gyengenek" latszott 150-180 PWM-en, holott a 4 pontos futas
//   szerint 1 %-on belul egyforma a ketto. Az utkozott meres NEM meres.)
//
//  A lista barmilyen hosszu lehet -- a program a TENYLEGES legnagyobb
//  PWM-mel szamol es azt is irja ki (regen fixen "240"-et irt, ami hazugsag
//  volt, ha rovidebb listat hasznaltal).
const int PWM_DB = 8;
int pwmLista[PWM_DB] = { 60, 80, 100, 120, 150, 180, 210, 240 };
int BEALL_MS = 400;    // ennyi ideig gyorsul be a PWM-re
int MERES_MS = 500;    // ezen az ablakon merjuk a sebesseget

// --------------------- elojel (a K0 EREDMENY sorabol) ---------------------
const int ENC_EL[2] = { 1, -1 };   // <<< robot 1: a jobb enkoder forditva szamol

// --------------------- eredmenyek ---------------------
float sebBalE[PWM_DB],  sebJobbE[PWM_DB];    // elore, impulzus/s
float sebBalH[PWM_DB],  sebJobbH[PWM_DB];    // hatra,  impulzus/s

// --------------------- egy sopres ---------------------
// irany: +1 = elore, -1 = hatra
void sopres(int irany, float kiBal[], float kiJobb[]) {
  for (int i = 0; i < PWM_DB; i++) {
    int p = pwmLista[i];

    motor(irany * p, irany * p);
    varj(BEALL_MS);                       // beall a sebesseg

    long b0 = encBal, j0 = encJobb;
    unsigned long t0 = millis();
    varj(MERES_MS);                       // ezen az ablakon merunk
    unsigned long dt = millis() - t0;
    long db = encBal - b0, dj = encJobb - j0;

    if (db < 0) db = -db;
    if (dj < 0) dj = -dj;

    kiBal[i]  = (float)db * 1000.0 / (float)dt;
    kiJobb[i] = (float)dj * 1000.0 / (float)dt;

    uzenet("PWM " + String(p) + " -> bal " + String(kiBal[i], 0)
           + " imp/s,  jobb " + String(kiJobb[i], 0) + " imp/s");
  }
  motor(0, 0);
  varj(600);
}

// --------------------- egyenes-illesztes ---------------------
// y = a + b*x  (x = sebesseg imp/s, y = PWM). Csak a mozgo pontokat hasznaljuk.
// a -> holtsav,  b -> KFF
void illesztes(float seb[], float &a, float &b) {
  float n = 0, sx = 0, sy = 0, sxy = 0, sxx = 0;
  for (int i = 0; i < PWM_DB; i++) {
    if (seb[i] < 30) continue;                 // meg all vagy csak rezeg: nem hasznalhato
    float x = seb[i], y = (float)pwmLista[i];
    n++; sx += x; sy += y; sxy += x * y; sxx += x * x;
  }
  if (n < 3) { a = 0; b = 0; return; }         // tul keves pont
  float nevezo = n * sxx - sx * sx;
  if (nevezo < 0.001) { a = 0; b = 0; return; }
  b = (n * sxy - sx * sy) / nevezo;
  a = (sy - b * sx) / n;
}

// --------------------- elojel-ellenorzes ---------------------
bool elojelRendben() {
  enkoderNullaz();
  varj(150);
  motor(90, 90);
  varj(450);
  motor(0, 0);
  varj(500);
  long b = encBal * ENC_EL[0], j = encJobb * ENC_EL[1];
  uzenet("Elojel-teszt: bal=" + String(b) + " jobb=" + String(j));
  motor(-90, -90); varj(450); motor(0, 0); varj(600);
  if (b < 50 || j < 50) {
    uzenet("=========== K3 LEALLT: ELOJEL-HIBA ===========");
    uzenet("Futtasd a K0-t, es az altala kiirt ENC_ELOJEL sort ird be az ENC_EL tombbe.");
    uzenet("=============================================");
    return false;
  }
  return true;
}

// --------------------- monotonitas-ellenorzes ---------------------
//  Nagyobb PWM-nek nagyobb sebesseget KELL adnia. Ha nem, a robot valamibe
//  utkozott, vagy elfogyott a hely, vagy valaki hozzaert -- a meres romlott.
bool monoton(float seb[], String nev) {
  bool jo = true;
  for (int i = 1; i < PWM_DB; i++) {
    if (seb[i] < seb[i-1] * 0.95) {
      uzenet("FIGYELEM (" + nev + "): PWM " + String(pwmLista[i]) + " LASSABB, mint PWM "
             + String(pwmLista[i-1]) + " (" + String(seb[i], 0) + " < " + String(seb[i-1], 0) + ")");
      jo = false;
    }
  }
  return jo;
}

void indulas() {
  uzenet("K3 -- PWM/sebesseg jelleggorbe.");
  uzenet("HELY: a 8 pontos lista TOBB MINT 5 m-t megy elore! Roviditsd, ha nincs annyi.");
  varj(1500);

  if (!elojelRendben()) return;

  uzenet("Indulas 3 mp mulva, NE nyulj a robothoz.");
  varj(3000);

  enkoderNullaz();
  uzenet("--- ELORE ---");
  sopres(+1, sebBalE, sebJobbE);
  varj(2000);
  uzenet("--- HATRA ---");
  sopres(-1, sebBalH, sebJobbH);

  // iranyonkenti illesztes
  float aBE, bBE, aJE, bJE, aBH, bBH, aJH, bJH;
  illesztes(sebBalE,  aBE, bBE);
  illesztes(sebJobbE, aJE, bJE);
  illesztes(sebBalH,  aBH, bBH);
  illesztes(sebJobbH, aJH, bJH);

  // a show-vezerles kerekenkent egy szammal dolgozik: a ket irany atlaga
  float holtBal  = (aBE + aBH) / 2.0;
  float holtJobb = (aJE + aJH) / 2.0;
  float kffBal   = (bBE + bBH) / 2.0;
  float kffJobb  = (bJE + bJH) / 2.0;

  // legnagyobb biztonsagos sebesseg: a leglassabb kerek/irany a csucs-PWM-en
  float vmax = sebBalE[PWM_DB - 1];
  if (sebJobbE[PWM_DB - 1] < vmax) vmax = sebJobbE[PWM_DB - 1];
  if (sebBalH[PWM_DB - 1]  < vmax) vmax = sebBalH[PWM_DB - 1];
  if (sebJobbH[PWM_DB - 1] < vmax) vmax = sebJobbH[PWM_DB - 1];

  motor(0, 0);

  // a meres minoseget ELOSZOR nezzuk meg -- romlott adatbol nincs ertelme szamolni
  uzenet("--- meres-ellenorzes ---");
  bool rendben = true;
  rendben &= monoton(sebBalE,  "bal elore");
  rendben &= monoton(sebJobbE, "jobb elore");
  rendben &= monoton(sebBalH,  "bal hatra");
  rendben &= monoton(sebJobbH, "jobb hatra");
  if (!rendben) {
    uzenet("A gorbe nem monoton: a robot valoszinuleg utkozott, vagy elfogyott a hely.");
    uzenet("EZ NEM MERES. Adj tobb helyet vagy roviditsd a pwmLista-t, es merj ujra.");
  }

  uzenet("=========== K3 EREDMENY ===========");

  // az illesztes akkor ad 0-t, ha 3-nal kevesebb hasznalhato pont volt
  if (kffBal < 0.0001 || kffJobb < 0.0001) {
    uzenet("!!! NEM SIKERULT AZ ILLESZTES -- NINCS EREDMENY !!!");
    uzenet("!!! A KFF es a HOLTSAV alabbi 0 erteke NEM MERES, hanem hibajelzes.");
    uzenet("!!! Az illeszteshez legalabb 3 olyan PWM-pont kell, ahol mozgott a kerek.");
    uzenet("!!! A pwmLista-ban most " + String(PWM_DB) + " pont van. Tegyel bele legalabb 4-et,");
    uzenet("!!! peldaul { 60, 80, 100, 120 }, es merj ujra.");
    uzenet("!!! SEMMIT ne irj at a tablazatban ez alapjan.");
    uzenet("===================================");
    return;
  }

  uzenet("--> KFF a tablazatba:     { " + String(kffBal, 5) + ", " + String(kffJobb, 5) + " }");
  uzenet("--> HOLTSAV (illesztett): { " + String(holtBal, 0) + ", " + String(holtJobb, 0) + " }");
  uzenet("    (hasonlitsd ossze a K0 holtsav-ertekeivel: 15 PWM-en belul legyenek!)");

  // a TENYLEGES legnagyobb PWM-et irjuk ki, nem fixen "240"-et
  int csucsPWM = pwmLista[PWM_DB - 1];
  uzenet("meg PWM " + String(csucsPWM) + "-nel: " + String(vmax, 0) + " imp/s a leglassabb kerek");
  uzenet("--> VMAX_IMPS a tablazatba (65 %): " + String(vmax * 0.65, 0));
  if (csucsPWM < 240)
    uzenet("    (a lista csak PWM " + String(csucsPWM)
           + "-ig ment, ezert ez OVATOS ertek -- 240-ig merve nagyobb lenne)");
  uzenet("    A show SOHA ne menjen ennel gyorsabban -- a maradek 35 % a PD-e,");
  uzenet("    hogy legyen mivel korrigalnia. Kifutott motorral nincs szabalyzas.");
  if (!rendben)
    uzenet("!!! De eloszor javitsd a fenti meres-hibat -- ezek a szamok romlott adatbol jottek.");
  uzenet("===================================");
}

void vezerles() {
  // Minden az indulas()-ban tortenik.
  motor(0, 0);
}
