// =====================================================================
//  K0 -- ELOJEL-ELLENORZES ES HOLTSAV-MERES
//  Show-koreografia: 1. meres a hat robotbol mindegyiken
// =====================================================================
//
//  MIT MER?
//    1. Jo iranyba szamol-e a ket enkoder, jo iranyba forog-e a ket motor.
//    2. HOLTSAV: az a legkisebb PWM, amitol a kerek MAR megmozdul.
//       Kerekenkent ES iranyonkent kulon, mert nem egyformak.
//
//  MIERT KELL?
//    A show-vezerles feed-forward tagja igy szamol:  PWM = HOLTSAV + u
//    Holtsav nelkul a lassu reszeknel (kis sebesseg) a kerek meg sem indul,
//    a robot "beragad", es a koreografiabol kiesik.
//
//  ELOKESZULET
//    - A robot a SHOW PADLOJAN alljon (nem bakon!): a holtsav terheles alatt
//      mas, mint szabadon porgo kereknel.
//    - Hagyj korulotte kb. 1 m szabad helyet: egy kerekkel hajtva pordul.
//    - Friss/teli akku. A holtsav akkufeszultseg-fuggo!
//    - Ird fel a Muszerfalon a V paranccsal az akkufeszultseget.
//
//  HASZNALAT
//    Beillesztes -> Fordit+Feltolt -> Start. Kb. 40 masodperc.
//    Az eredmenyt a naplo utolso soraibol masold at a robot_adatok.md-be.
// =====================================================================

// --------------------- beallitasok ---------------------
int hMin      = 25;    // innen indul a PWM-sopres
int hMax      = 140;   // eddig megy (ha eddig sem mozdul: mechanikai hiba!)
int hLepes    = 5;     // PWM-lepeskoz
int hIdo      = 220;   // ms egy lepesben
int hKuszob   = 4;     // ennyi impulzus mar "megmozdult"

// --------------------- segedfuggveny ---------------------
// oldal: 0 = bal, 1 = jobb    irany: +1 = elore, -1 = hatra
int holtsavMeres(int oldal, int irany) {
  for (int p = hMin; p <= hMax; p += hLepes) {
    long elotte = (oldal == 0) ? encBal : encJobb;

    if (oldal == 0) motor(irany * p, 0);
    else            motor(0, irany * p);
    varj(hIdo);

    long utana = (oldal == 0) ? encBal : encJobb;
    motor(0, 0);
    varj(120);

    long valtozas = utana - elotte;
    if (valtozas < 0) valtozas = -valtozas;
    if (valtozas >= hKuszob) return p;
  }
  return -1;   // nem mozdult meg sehol
}

void indulas() {
  uzenet("K0 -- elojel es holtsav. NE nyulj a robothoz!");
  varj(1500);

  // ============ 1. ELOJEL-ELLENORZES ============
  enkoderNullaz();
  varj(200);
  motor(90, 90);
  varj(500);
  motor(0, 0);
  varj(500);

  uzenet("1) ELORE hajtva:  encBal=" + String(encBal) + "  encJobb=" + String(encJobb));
  uzenet("   Mindkettonek POZITIVNAK kell lennie, es a robotnak ELORE kell mennie.");
  uzenet("   Ha nem: Muszerfal -> Robot beallitasa -> 'Enkoder-irany felismerese' (N,AUTO),");
  uzenet("   utana futtasd ujra ezt a merest. A kodot NE ird at.");

  // vissza a kiindulasi hely kozelebe
  motor(-90, -90);
  varj(500);
  motor(0, 0);
  varj(800);

  // ============ 2. HOLTSAV ============
  uzenet("2) Holtsav-meres indul (4 x kb. 8 mp)...");

  int hBalElore = holtsavMeres(0, +1);
  varj(400);
  int hBalHatra = holtsavMeres(0, -1);
  varj(400);
  int hJobbElore = holtsavMeres(1, +1);
  varj(400);
  int hJobbHatra = holtsavMeres(1, -1);
  varj(400);

  motor(0, 0);

  uzenet("=========== K0 EREDMENY ===========");
  uzenet("HOLTSAV bal  elore/hatra: " + String(hBalElore) + " / " + String(hBalHatra));
  uzenet("HOLTSAV jobb elore/hatra: " + String(hJobbElore) + " / " + String(hJobbHatra));

  // a show-vezerles kerekenkent EGY szamot hasznal: a ket irany atlaga
  if (hBalElore > 0 && hBalHatra > 0 && hJobbElore > 0 && hJobbHatra > 0) {
    int hBal  = (hBalElore + hBalHatra) / 2;
    int hJobb = (hJobbElore + hJobbHatra) / 2;
    uzenet("--> HOLTSAV a tablazatba:  { " + String(hBal) + ", " + String(hJobb) + " }");
    if (hBal > 100 || hJobb > 100)
      uzenet("FIGYELEM: 100 folotti holtsav = szorulo kerek, gyenge akku vagy laza tengely!");
    int kulonbseg = hBal - hJobb;
    if (kulonbseg < 0) kulonbseg = -kulonbseg;
    if (kulonbseg > 25)
      uzenet("FIGYELEM: a ket kerek holtsava nagyon elter (" + String(kulonbseg) + " PWM) -- nezd meg a mechanikat.");
  } else {
    uzenet("HIBA: valamelyik kerek " + String(hMax) + " PWM-ig sem mozdult meg.");
    uzenet("Ellenorizd: STBY laba HIGH? motor kabel? akku? szorul a kerek?");
  }
  uzenet("===================================");
  uzenet("Ird be a robot_adatok.md tablazatba, majd jon a K1.");
}

void vezerles() {
  // Minden az indulas()-ban tortenik, itt nincs teendo.
  motor(0, 0);
}
